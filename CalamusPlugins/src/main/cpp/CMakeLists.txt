cmake_minimum_required(VERSION 3.22.1)
project(calamus_python C)
set(CMAKE_C_STANDARD 11)
set(PYTHON_ROOT ${CMAKE_CURRENT_LIST_DIR}/../../../vendor/python311)
add_library(python311 SHARED IMPORTED)
set_target_properties(python311 PROPERTIES IMPORTED_NO_SONAME TRUE IMPORTED_LOCATION ${PYTHON_ROOT}/jniLibs/${ANDROID_ABI}/libpython3.11.so)
add_library(calamus_python SHARED runtime.c)
target_include_directories(calamus_python PRIVATE ${PYTHON_ROOT}/${ANDROID_ABI}/include/python3.11)
target_compile_options(calamus_python PRIVATE -fvisibility=hidden -Wall -Wextra -Werror=return-type)
target_link_options(calamus_python PRIVATE -Wl,-z,max-page-size=16384)
target_link_libraries(calamus_python python311 android log)

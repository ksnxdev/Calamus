cmake_minimum_required(VERSION 3.22.1)
project(calamus_extera_hook C CXX)
if(ANDROID_ABI STREQUAL "arm64-v8a" OR ANDROID_ABI STREQUAL "armeabi-v7a")
    find_package(exterahook REQUIRED CONFIG)
    find_package(lsplant REQUIRED CONFIG)
    find_package(shadowhook REQUIRED CONFIG)
    add_library(extera_hook_runtime SHARED bootstrap.c)
    set_target_properties(extera_hook_runtime PROPERTIES OUTPUT_NAME exterahook LINKER_LANGUAGE CXX)
    target_link_libraries(extera_hook_runtime PRIVATE exterahook::exterahook lsplant::lsplant lsplant::dex_builder_static shadowhook::shadowhook android log dl z)
else()
    add_library(extera_hook_runtime SHARED unavailable.c)
    set_target_properties(extera_hook_runtime PROPERTIES OUTPUT_NAME exterahook LINKER_LANGUAGE CXX)
endif()
target_link_options(extera_hook_runtime PRIVATE -Wl,-z,max-page-size=16384)
